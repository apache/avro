29/// Main entry point that dispatches to the appropriate architecture
30///
31/// # Safety
32/// May use native CPU features
33#[inline(always)]
34#[cfg(target_arch = "aarch64")]
35pub(crate) unsafe fn update(state: u64, bytes: &[u8], params: &CrcParams) -> u64 {
36    use crate::feature_detection::{get_arch_ops, ArchOpsInstance};
37
38    match get_arch_ops() {
39        ArchOpsInstance::Aarch64AesSha3(ops) => update_aarch64_aes_sha3(state, bytes, params, *ops),
40        ArchOpsInstance::Aarch64Aes(ops) => update_aarch64_aes(state, bytes, params, *ops),
41        ArchOpsInstance::SoftwareFallback => {
42            #[cfg(feature = "std")]
43            let has_features =
44                is_aarch64_feature_detected!("aes") && is_aarch64_feature_detected!("neon");
45            #[cfg(not(feature = "std"))]
46            let has_features = cfg!(target_feature = "aes") && cfg!(target_feature = "neon");
47
48            if !has_features {
49                #[cfg(any(not(target_feature = "aes"), not(target_feature = "neon")))]
50                {
51                    // Use software implementation when no SIMD support is available
52                    return crate::arch::software::update(state, bytes, params);
53                }
54            }
55
56            // This should likely never happen, but just in case
57            panic!("aarch64 features missing (NEON and/or AES)");
58        }
59    }
60}

62#[inline]
63#[cfg(target_arch = "aarch64")]
64#[target_feature(enable = "aes")]
65unsafe fn update_aarch64_aes(
66    state: u64,
67    bytes: &[u8],
68    params: &CrcParams,
69    ops: Aarch64AesOps,
70) -> u64 {
71    match params.width {
72        64 => algorithm::update::<_, Width64>(state, bytes, params, &ops),
73        32 => algorithm::update::<_, Width32>(state as u32, bytes, params, &ops) as u64,
74        16 => algorithm::update::<_, Width16>(state as u16, bytes, params, &ops) as u64,
75        _ => panic!("Unsupported CRC width: {}", params.width),
76    }
77}

96/// Main entry point for x86/x86_64
97///
98/// # Safety
99/// May use native CPU features
100#[inline(always)]
101#[cfg(any(target_arch = "x86", target_arch = "x86_64"))]
102pub(crate) unsafe fn update(state: u64, bytes: &[u8], params: &CrcParams) -> u64 {
103    use crate::feature_detection::{get_arch_ops, ArchOpsInstance};
104
105    match get_arch_ops() {
106        #[cfg(target_arch = "x86_64")]
107        ArchOpsInstance::X86_64Avx512Vpclmulqdq(ops) => {
108            update_x86_64_avx512_vpclmulqdq(state, bytes, params, *ops)
109        }
110        #[cfg(target_arch = "x86_64")]
111        ArchOpsInstance::X86_64Avx512Pclmulqdq(ops) => {
112            update_x86_64_avx512_pclmulqdq(state, bytes, params, *ops)
113        }
114        ArchOpsInstance::X86SsePclmulqdq(ops) => {
115            update_x86_sse_pclmulqdq(state, bytes, params, *ops)
116        }
117        ArchOpsInstance::SoftwareFallback => crate::arch::software::update(state, bytes, params),
118    }
119}

300            (&[0; 4096], 0x6482d367eb22b64e),

303            // custom values

330    /// Test the "crc32" variant used in PHP's hash() function, which is different from the
331    /// crc32() function. It's really just CRC-32/BZIP2 with the output byte-reversed to little
332    /// endian.
333    ///
334    /// https://www.php.net/manual/en/function.hash-file.php#104836
335    #[test]
336    fn test_crc32_php_standard_vectors() {
337        static CASES: &[(&[u8], u64)] = &[
338            (b"123456789", 0x181989fc),
339            (&[0; 4096], 0xe3380088),
340            (&[255; 4096], 0x8f2ae650),
341            (b"hello, world!", 0x5eacce7),
342        ];
343
344        for (input, expected) in CASES {
345            let bzip2_crc = unsafe {
346                (update(CRC32_BZIP2.init, input, &CRC32_BZIP2) ^ CRC32_BZIP2.xorout) as u32
347            };
348
349            // PHP reverses the byte order of the CRC for some reason
350            let actual = bzip2_crc.swap_bytes();
351
352            assert_eq!(
353                actual, *expected as u32,
354                "Mismatch CRC, expected {:#x}, got {:#x}, input: {:?}",
355                expected, actual, input
356            );
357        }
358    }

434    fn test_length(length: usize, config: &AnyCrcTestConfig) {
435        let mut data = vec![0u8; length];
436        rng().fill(&mut data[..]);
437
438        // Calculate expected CRC using the reference implementation
439        let expected = config.checksum_with_reference(&data);
440
441        // direct update() call, which needs XOROUT applied
442        let actual = unsafe {
443            update(config.get_init_algorithm(), &data, config.get_params()) ^ config.get_xorout()
444        };
445
446        assert_eq!(
447            actual,
448            expected,
449            "\nFailed for {} with length {}\nGot: {:016x}\nExpected: {:016x}",
450            config.get_name(),
451            length,
452            actual,
453            expected
454        );
455    }
