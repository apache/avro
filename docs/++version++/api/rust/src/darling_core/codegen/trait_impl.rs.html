24impl<'a> TraitImpl<'a> {
25    /// Get all declared type parameters.
26    pub fn declared_type_params(&self) -> IdentSet {
27        self.generics
28            .type_params()
29            .map(|tp| tp.ident.clone())
30            .collect()
31    }
32
33    /// Get the type parameters which are used by non-skipped, non-magic fields.
34    /// These type parameters will have a `FromMeta` bound applied to them in emitted
35    /// code.
36    pub fn used_type_params(&self) -> IdentSet {
37        self.type_params_matching(|f| !f.skip, |v| !v.skip)
38    }
39
40    fn type_params_matching<F, V>(&self, field_filter: F, variant_filter: V) -> IdentSet
41    where
42        F: Fn(&&Field<'_>) -> bool,
43        V: Fn(&&Variant<'_>) -> bool,
44    {
45        let declared = self.declared_type_params();
46        match self.data {
47            Data::Struct(ref v) => self.type_params_in_fields(v, &field_filter, &declared),
48            Data::Enum(ref v) => {
49                v.iter()
50                    .filter(variant_filter)
51                    .fold(Default::default(), |mut state, variant| {
52                        state.extend(self.type_params_in_fields(
53                            &variant.data,
54                            &field_filter,
55                            &declared,
56                        ));
57                        state
58                    })
59            }
60        }
61    }
62
63    /// Get the type parameters of all fields in a set matching some filter
64    fn type_params_in_fields<'b, F>(
65        &'b self,
66        fields: &'b Fields<Field<'a>>,
67        field_filter: F,
68        declared: &IdentSet,
69    ) -> IdentSet
70    where
71        F: Fn(&&'b Field<'_>) -> bool,
72    {
73        fields
74            .iter()
75            .filter(field_filter)
76            .collect_type_params_cloned(&Purpose::BoundImpl.into(), declared)
77    }
78}

80impl<'a> TraitImpl<'a> {
81    /// Gets the `let` declaration for errors accumulated during parsing.
82    pub fn declare_errors(&self) -> ErrorDeclaration {
83        ErrorDeclaration::default()
84    }
85
86    /// Gets the check which performs an early return if errors occurred during parsing.
87    pub fn check_errors(&self) -> ErrorCheck<'_> {
88        ErrorCheck::default()
89    }
90
91    /// Generate local variable declarations for all fields.
92    pub(in crate::codegen) fn local_declarations(&self) -> TokenStream {
93        if let Data::Struct(ref vd) = self.data {
94            let vdr = vd.as_ref().map(Field::as_declaration);
95            let decls = vdr.fields.as_slice();
96            quote!(#(#decls)*)
97        } else {
98            quote!()
99        }
100    }
101
102    pub(in crate::codegen) fn post_transform_call(&self) -> Option<TokenStream> {
103        self.post_transform.map(|pt| quote!(#pt))
104    }
105
106    /// Generate local variable declaration and initialization for instance from which missing fields will be taken.
107    pub(in crate::codegen) fn fallback_decl(&self) -> TokenStream {
108        let default = self.default.as_ref().map(DefaultExpression::as_declaration);
109        quote!(#default)
110    }
111
112    pub fn require_fields(&self) -> TokenStream {
113        if let Data::Struct(ref vd) = self.data {
114            let check_nones = vd.as_ref().map(Field::as_presence_check);
115            let checks = check_nones.fields.as_slice();
116
117            // If a field was marked `flatten`, now is the time to process any unclaimed meta items
118            // and mark the field as having been seen.
119            let flatten_field_init = vd.fields.iter().find(|f| f.flatten).map(|v| {
120                v.as_flatten_initializer(vd.fields.iter().filter_map(Field::as_name).collect())
121            });
122
123            quote! {
124                #flatten_field_init
125                #(#checks)*
126            }
127        } else {
128            quote!()
129        }
130    }
131
132    pub(in crate::codegen) fn initializers(&self) -> TokenStream {
133        self.make_field_ctx().initializers()
134    }
135
136    /// Generate the loop which walks meta items looking for property matches.
137    pub(in crate::codegen) fn core_loop(&self) -> TokenStream {
138        self.make_field_ctx().core_loop()
139    }
140
141    fn make_field_ctx(&'a self) -> FieldsGen<'a> {
142        match self.data {
143            Data::Enum(_) => panic!("Core loop on enums isn't supported"),
144            Data::Struct(ref data) => FieldsGen::new(data, self.allow_unknown_fields),
145        }
146    }
147}
