1//! Struct is "transparent" (delegates impl to inner field) when there
2//! is just a single field. If this field is unnamed, the struct is transparent by default
3
4use proc_macro2::Span;
5use syn::{Index, Member};
6
7use crate::{
8    ast::{Fields, Style},
9    codegen::Field,
10};
11
12/// If this struct is "transparent", get the member + inner field
13pub fn extract_transparent<'a, 'b>(
14    fields: &'a Fields<Field<'b>>,
15    is_transparent: bool,
16) -> Option<(Member, &'a Field<'b>)> {
17    if fields.len() != 1 {
18        return None;
19    }
20    let field = fields
21        .iter()
22        .next()
23        .expect("returned earlier on `.len() != 1`");
24
25    match fields.style {
26        Style::Tuple => Some((
27            Member::Unnamed(Index {
28                index: 0,
29                span: Span::call_site(),
30            }),
31            field,
32        )),
33        Style::Struct if is_transparent => Some((Member::Named(field.ident.clone()), field)),
34        _ => None,
35    }
36}
