18use crate::attributes::{NamedTypeOptions, VariantOptions};
19use crate::enums::{SchemaType, variant_to_schema_expr};
20use crate::implementation::Implementation;
21use crate::utils::json_value_expr;
22use proc_macro2::Ident;
23use quote::quote;
24use std::collections::HashSet;
25use syn::spanned::Spanned;
26use syn::{DataEnum, Generics};
27
28pub fn to_implementation(
29    ident: Ident,
30    generics: Generics,
31    container_attrs: NamedTypeOptions,
32    data: DataEnum,
33) -> Result<Implementation, Vec<syn::Error>> {
34    let mut errors = Vec::new();
35    let mut variant_exprs = Vec::new();
36
37    let mut names = HashSet::new();
38
39    for variant in data.variants {
40        let variant_attrs = match VariantOptions::new(&variant.attrs, variant.span()) {
41            Ok(attrs) => attrs,
42            Err(errs) => {
43                errors.extend(errs);
44                continue;
45            }
46        };
47
48        if variant_attrs.skip {
49            continue;
50        }
51
52        match variant_to_schema_expr(
53            variant,
54            variant_attrs,
55            container_attrs.rename_all,
56            container_attrs.rename_all_fields,
57            false,
58            false,
59            |info| {
60                if let SchemaType::Named(name) = info.schema_type
61                    && !names.insert(name.to_string())
62                {
63                    Err(format!(
64                        "AvroSchema: Duplicate variant names detected: {name}"
65                    ))
66                } else {
67                    Ok(())
68                }
69            },
70        ) {
71            Ok(expr) => variant_exprs.push(expr),
72            Err(errs) => errors.extend(errs),
73        }
74    }
75
76    if !errors.is_empty() {
77        return Err(errors);
78    }
79
80    let schema_expr = quote! {{
81        let mut builder = ::apache_avro::schema::UnionSchema::builder();
82
83        #(builder.variant(#variant_exprs).expect("Duplicate names found");)*
84
85        ::apache_avro::schema::Schema::Union(builder.build())
86    }};
87
88    Ok(Implementation::unnamed(
89        ident,
90        generics,
91        schema_expr,
92        None,
93        container_attrs
94            .default
95            .map(json_value_expr)
96            .map(|t| quote! { ::std::option::Option::Some(#t)}),
97    ))
98}
