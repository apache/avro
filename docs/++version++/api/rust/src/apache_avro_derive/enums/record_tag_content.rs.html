18use crate::attributes::{NamedTypeOptions, Repr, VariantOptions};
19use crate::case::RenameRule;
20use crate::enums::{FieldVariants, variant_to_schema_expr};
21use crate::implementation::Implementation;
22use crate::utils::{aliases, json_value_expr, name_expr, preserve_optional, rename_ident};
23use proc_macro2::{Ident, Span};
24use quote::quote;
25use std::collections::HashSet;
26use syn::spanned::Spanned;
27use syn::{DataEnum, Generics};
28
29pub fn to_implementation(
30    input_span: Span,
31    ident: Ident,
32    generics: Generics,
33    container_attrs: NamedTypeOptions,
34    data: DataEnum,
35) -> Result<Implementation, Vec<syn::Error>> {
36    let Some(Repr::RecordTagContent { tag, content }) = container_attrs.repr else {
37        unreachable!()
38    };
39    let mut errors = Vec::new();
40    let mut symbols = Vec::new();
41    let mut variant_exprs = Vec::new();
42
43    for variant in data.variants {
44        let variant_attrs = match VariantOptions::new(&variant.attrs, variant.span()) {
45            Ok(attrs) => attrs,
46            Err(errs) => {
47                errors.extend(errs);
48                continue;
49            }
50        };
51
52        if variant_attrs.skip {
53            continue;
54        }
55
56        let name = rename_ident(
57            &variant.ident,
58            variant_attrs.rename.clone(),
59            container_attrs.rename_all,
60            RenameRule::apply_to_variant,
61        );
62
63        symbols.push(name);
64
65        match variant_to_schema_expr(
66            variant,
67            variant_attrs,
68            container_attrs.rename_all,
69            container_attrs.rename_all_fields,
70            true,
71            true,
72            |info| {
73                match info.variant {
74                FieldVariants::Unnamed(0) => Err(r#"AvroSchema: Empty tuple variants are not supported for `#[serde(tag = "..", content = "..")]` enums"#.into()),
75                _ => Ok(())
76            }
77            },
78        ) {
79            Ok(expr) => variant_exprs.push(expr),
80            Err(errs) => errors.extend(errs),
81        }
82    }
83
84    if symbols.iter().collect::<HashSet<_>>().len() != symbols.len() {
85        errors.push(syn::Error::new(
86            input_span,
87            format!("AvroSchema: Duplicate variant names detected: {symbols:?}"),
88        ));
89    }
90
91    if !errors.is_empty() {
92        return Err(errors);
93    }
94
95    let doc = preserve_optional(
96        container_attrs
97            .doc
98            .as_ref()
99            .map(|s| quote! { #s.to_string() }),
100    );
101    let enum_aliases = aliases(&container_attrs.aliases);
102    let tag_name_expr = name_expr(&tag);
103
104    let schema_expr = quote! {{
105        let mut builder = ::apache_avro::schema::UnionSchema::builder();
106        #(builder.variant_ignore_duplicates(#variant_exprs).expect("Got two map or arrays with different inner types");)*
107
108        let content_contains_null = builder.contains(&::apache_avro::schema::Schema::Null);
109        let content_schema = ::apache_avro::schema::Schema::Union(builder.build());
110
111        let tag_schema = ::apache_avro::schema::Schema::r#enum(#tag_name_expr, vec![#(#symbols.to_owned()),*]).build();
112
113        let mut fields = ::std::vec::Vec::with_capacity(2);
114        fields.push(::apache_avro::schema::RecordField::builder()
115            .name(#tag)
116            .schema(tag_schema)
117            .build()
118        );
119        fields.push(::apache_avro::schema::RecordField::builder()
120            .name(#content)
121            .schema(content_schema)
122            // This is needed for unit variants, for which Serde won't serialize content
123            .maybe_default(if content_contains_null { Some(::serde_json::Value::Null) } else { None })
124            .build()
125        );
126        ::apache_avro::schema::Schema::Record(::apache_avro::schema::RecordSchema::builder()
127            .name(name)
128            .aliases(#enum_aliases)
129            .doc(#doc)
130            .fields(fields)
131            .build()
132        )
133    }};
134
135    let record_fields_expr = quote! {{
136        let mut builder = ::apache_avro::schema::UnionSchema::builder();
137        #(builder.variant_ignore_duplicates(#variant_exprs).expect("Got two map or arrays with different inner types");)*
138
139        let content_contains_null = builder.contains(&::apache_avro::schema::Schema::Null);
140        let content_schema = ::apache_avro::schema::Schema::Union(builder.build());
141
142        let tag_schema = ::apache_avro::schema::Schema::r#enum(#tag_name_expr, vec![#(#symbols.to_owned()),*]).build();
143
144        let mut fields = ::std::vec::Vec::with_capacity(2);
145        fields.push(::apache_avro::schema::RecordField::builder()
146            .name(#tag)
147            .schema(tag_schema)
148            .build()
149        );
150        fields.push(::apache_avro::schema::RecordField::builder()
151            .name(#content)
152            .schema(content_schema)
153            // This is needed for unit variants, for which Serde won't serialize content
154            .maybe_default(if content_contains_null { Some(::serde_json::Value::Null) } else { None })
155            .build()
156        );
157
158        Some(fields)
159    }};
160
161    Ok(Implementation::named(
162        ident,
163        generics,
164        &container_attrs.name,
165        schema_expr,
166        Some(record_fields_expr),
167        container_attrs
168            .default
169            .map(json_value_expr)
170            .map(|t| quote! { ::std::option::Option::Some(#t)}),
171    ))
172}
