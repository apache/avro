1//! Rust binding to the [zstd library][zstd].
2//!
3//! This crate provides:
4//!
5//! * An [encoder](stream/write/struct.Encoder.html) to compress data using zstd
6//!   and send the output to another write.
7//! * A [decoder](stream/read/struct.Decoder.html) to read input data from a `Read`
8//!   and decompress it.
9//! * Convenient functions for common tasks.
10//!
11//! # Example
12//!
13//! ```no_run
14//! use std::io;
15//!
16//! // Uncompress input and print the result.
17//! zstd::stream::copy_decode(io::stdin(), io::stdout()).unwrap();
18//! ```
19//!
20//! # Cargo features
21//!
22//! Enabled by default: `legacy`, `arrays`, `zdict_builder`.
23//!
24//! | Feature | Default | Description |
25//! | --- | --- | --- |
26//! | `arrays` | yes | Use fixed-size arrays (`[u8; N]`) as output buffers. |
27//! | `legacy` | yes | Decode frames written by zstd versions older than 0.8. |
28//! | `zdict_builder` | yes | Train new dictionaries. *Using* a dictionary always works. |
29//! | `experimental` | | Expose zstd's experimental API, which has no stability guarantees. |
30//! | `zstdmt` | | Multi-threaded compression inside the C library. |
31//! | `thin` | | Build a smaller C library, at some cost in speed and error reporting. |
32//! | `debug` | | Enable zstd's debug logs. |
33//! | `no_asm` | | Do not build the x86-64 assembly. |
34//! | `fat-lto`, `thin-lto` | | Cross-language LTO. Only works if `clang` builds the C library. |
35//! | `bindgen` | | Generate the bindings at build time rather than using the pre-generated ones. |
36//! | `pkg-config` | | Link against a system-installed libzstd instead of building the bundled source. |
37//! | `vendored` | | Always build the bundled source, even when `pkg-config` is also enabled. |
38//! | `cmake` | | Build the C library with zstd's own CMake files instead of the `cc` crate. |
39//! | `rust-backend` | | Use [libzstd-rs-sys], a Rust port of zstd, instead of the C library. Experimental. |
40//!
41//! See the [readme] for more about the build-related ones.
42//!
43//! [zstd]: https://github.com/facebook/zstd
44//! [readme]: https://github.com/gyscos/zstd-rs#cargo-features
45//! [libzstd-rs-sys]: https://github.com/trifectatechfoundation/libzstd-rs-sys

84/// An error reported by the zstd library.
85///
86/// When zstd itself fails, the `io::Error` this crate returns carries one of
87/// these, so you can tell what went wrong without matching on the message:
88///
89/// ```
90/// let mut small = [0u8; 4];
91/// let err = zstd::bulk::compress_to_buffer(&[42u8; 1000], &mut small[..], 3)
92///     .unwrap_err();
93/// let zstd_err = err
94///     .get_ref()
95///     .and_then(|e| e.downcast_ref::<zstd::Error>())
96///     .unwrap();
97/// use zstd::zstd_safe::zstd_sys::ZSTD_ErrorCode;
98/// assert!(zstd_err.is(ZSTD_ErrorCode::ZSTD_error_dstSize_tooSmall));
99/// ```
100#[derive(Clone, Copy, PartialEq, Eq)]
101pub struct Error {
102    code: zstd_safe::ErrorCode,
103}
104
105impl Error {
106    /// Returns whether this is an error of the given kind.
107    ///
108    /// See [`zstd_safe::is_error_kind()`].
109    pub fn is(&self, kind: zstd_safe::zstd_sys::ZSTD_ErrorCode) -> bool {
110        zstd_safe::is_error_kind(self.code, kind)
111    }
112}
113
114impl std::fmt::Debug for Error {
115    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
116        f.debug_tuple("Error")
117            .field(&zstd_safe::get_error_name(self.code))
118            .finish()
119    }
120}
121
122impl std::fmt::Display for Error {
123    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
124        f.write_str(zstd_safe::get_error_name(self.code))
125    }
126}

135/// Returns an empty `Vec` with room for exactly `capacity` bytes.
136///
137/// Unlike `Vec::with_capacity`, this fails with an `OutOfMemory` error instead
138/// of aborting the process when that much memory cannot be allocated.
139fn vec_with_capacity(capacity: usize) -> io::Result<Vec<u8>> {
140    let mut vec = Vec::new();
141    vec.try_reserve_exact(capacity).map_err(|_| {
142        io::Error::new(
143            io::ErrorKind::OutOfMemory,
144            format!("could not allocate a buffer of {} bytes", capacity),
145        )
146    })?;
147    Ok(vec)
148}

152#[cfg(test)]
153fn test_cycle<F, G>(data: &[u8], f: F, g: G)
154where
155    F: Fn(&[u8]) -> Vec<u8>,
156    G: Fn(&[u8]) -> Vec<u8>,
157{
158    let mid = f(data);
159    let end = g(&mid);
160    assert_eq!(data, &end[..]);
161}

165where
166    F: Fn(&[u8]) -> io::Result<Vec<u8>>,
167    G: Fn(&[u8]) -> io::Result<Vec<u8>>,
168{
169    test_cycle(data, |data| f(data).unwrap(), |data| g(data).unwrap())
170}

172/// Compressible data, `len` bytes, different for each seed.
173///
174/// The worker thread tests use it in the sizes that keep a job busy long
175/// enough to be caught running.
176#[cfg(all(test, feature = "zstdmt"))]
177fn test_words(len: usize, seed: u64) -> Vec<u8> {
178    let words: [&[u8]; 6] =
179        [b"lorem ", b"ipsum ", b"dolor ", b"sit ", b"amet ", b"elit "];
180    let mut x = seed | 1;
181    let mut data = Vec::with_capacity(len + 8);
182    while data.len() < len {
183        x ^= x << 13;
184        x ^= x >> 7;
185        x ^= x << 17;
186        data.extend_from_slice(words[(x % 6) as usize]);
187    }
188    data.truncate(len);
189    data
190}
