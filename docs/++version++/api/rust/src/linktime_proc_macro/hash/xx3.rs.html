56#[inline]
57fn mul128_fold64(lhs: u64, rhs: u64) -> u64 {
58    let p = u128::from(lhs) * u128::from(rhs);
59    (p as u64) ^ ((p >> 64) as u64)
60}

62#[inline]
63fn avalanche64(mut h: u64) -> u64 {
64    h ^= h >> 33;
65    h = h.wrapping_mul(P64_2);
66    h ^= h >> 29;
67    h = h.wrapping_mul(P64_3);
68    h ^= h >> 32;
69    h
70}

72#[inline]
73fn avalanche3(mut h: u64) -> u64 {
74    h ^= h >> 37;
75    h = h.wrapping_mul(PRIME_MX1);
76    h ^= h >> 32;
77    h
78}

80#[inline]
81fn rrmxmx(mut h: u64, len: u64) -> u64 {
82    h ^= rotl64(h, 49) ^ rotl64(h, 24);
83    h = h.wrapping_mul(PRIME_MX2);
84    h ^= (h >> 35).wrapping_add(len);
85    h = h.wrapping_mul(PRIME_MX2);
86    h ^ (h >> 28)
87}

89fn len_1to3(input: &[u8], len: usize, secret: &[u8], seed: u64) -> u64 {
90    let c1 = input[0] as u32;
91    let c2 = input[len >> 1] as u32;
92    let c3 = input[len - 1] as u32;
93    let combined = (c1 << 16) | (c2 << 24) | c3 | ((len as u32) << 8);
94    let bitflip = u64::from(read_le32(secret, 0) ^ read_le32(secret, 4)).wrapping_add(seed);
95    avalanche64(u64::from(combined) ^ bitflip)
96}

98fn len_4to8(input: &[u8], len: usize, secret: &[u8], mut seed: u64) -> u64 {
99    seed ^= u64::from((seed as u32).swap_bytes()) << 32;
100    let input1 = u64::from(read_le32(input, 0));
101    let input2 = u64::from(read_le32(input, len - 4));
102    let bitflip = (read_le64(secret, 8) ^ read_le64(secret, 16)).wrapping_sub(seed);
103    let input64 = input2 + (input1 << 32);
104    rrmxmx(input64 ^ bitflip, len as u64)
105}

107fn len_9to16(input: &[u8], len: usize, secret: &[u8], seed: u64) -> u64 {
108    let bitflip1 = (read_le64(secret, 24) ^ read_le64(secret, 32)).wrapping_add(seed);
109    let bitflip2 = (read_le64(secret, 40) ^ read_le64(secret, 48)).wrapping_sub(seed);
110    let input_lo = read_le64(input, 0) ^ bitflip1;
111    let input_hi = read_le64(input, len - 8) ^ bitflip2;
112    let acc = (len as u64)
113        .wrapping_add(input_lo.swap_bytes())
114        .wrapping_add(input_hi)
115        .wrapping_add(mul128_fold64(input_lo, input_hi));
116    avalanche3(acc)
117}

119fn len_0to16(input: &[u8], len: usize, secret: &[u8], seed: u64) -> u64 {
120    if len > 8 {
121        return len_9to16(input, len, secret, seed);
122    }
123    if len >= 4 {
124        return len_4to8(input, len, secret, seed);
125    }
126    if len > 0 {
127        return len_1to3(input, len, secret, seed);
128    }
129    avalanche64(seed ^ (read_le64(secret, 56) ^ read_le64(secret, 64)))
130}

132#[inline]
133fn mix16b(input: &[u8], secret: &[u8], seed64: u64) -> u64 {
134    let il = read_le64(input, 0);
135    let ih = read_le64(input, 8);
136    mul128_fold64(
137        il ^ read_le64(secret, 0).wrapping_add(seed64),
138        ih ^ read_le64(secret, 8).wrapping_sub(seed64),
139    )
140}

142fn len_17to128(input: &[u8], len: usize, secret: &[u8], seed: u64) -> u64 {
143    let mut acc = (len as u64).wrapping_mul(P64_1);
144    let mut i = (len - 1) / 32;
145    loop {
146        acc = acc.wrapping_add(mix16b(&input[16 * i..], &secret[32 * i..], seed));
147        acc = acc.wrapping_add(mix16b(
148            &input[len - 16 * (i + 1)..],
149            &secret[32 * i + 16..],
150            seed,
151        ));
152        if i == 0 {
153            break;
154        }
155        i -= 1;
156    }
157    avalanche3(acc)
158}

160fn len_129to240(input: &[u8], len: usize, secret: &[u8], seed: u64) -> u64 {
161    let mut acc = (len as u64).wrapping_mul(P64_1);
162    let nb_rounds = len / 16;
163    for i in 0..8 {
164        acc = acc.wrapping_add(mix16b(&input[16 * i..], &secret[16 * i..], seed));
165    }
166    let off_last = SECRET_MIN - MIDSIZE_LASTOFF;
167    let mut acc_end = mix16b(&input[len - 16..], &secret[off_last..], seed);
168    acc = avalanche3(acc);
169    for i in 8..nb_rounds {
170        acc_end = acc_end.wrapping_add(mix16b(
171            &input[16 * i..],
172            &secret[16 * (i - 8) + MIDSIZE_START..],
173            seed,
174        ));
175    }
176    avalanche3(acc.wrapping_add(acc_end))
177}

181    u64::from(lhs as u32)
182        .wrapping_mul(u64::from(rhs as u32))
183        .wrapping_add(acc)

186fn accumulate_512(acc: &mut [u64; 8], input: &[u8], secret: &[u8]) {
187    for lane in 0..8 {
188        let data_val = read_le64(input, lane * 8);
189        let data_key = data_val ^ read_le64(secret, lane * 8);
190        acc[lane ^ 1] = acc[lane ^ 1].wrapping_add(data_val);
191        acc[lane] = mult32to64_add64(data_key, data_key >> 32, acc[lane]);
192    }
193}

195fn scramble_acc(acc: &mut [u64; 8], secret: &[u8]) {
196    for lane in 0..8 {
197        let key64 = read_le64(secret, lane * 8);
198        let mut a = acc[lane];
199        a ^= a >> 47;
200        a ^= key64;
201        a = a.wrapping_mul(P32_1);
202        acc[lane] = a;
203    }
204}

206fn accumulate(acc: &mut [u64; 8], input: &[u8], secret: &[u8], nb_stripes: usize) {
207    for n in 0..nb_stripes {
208        accumulate_512(
209            acc,
210            &input[n * STRIPE_LEN..],
211            &secret[n * SECRET_CONSUME_RATE..],
212        );
213    }
214}

216fn hash_long_internal_loop(
217    acc: &mut [u64; 8],
218    input: &[u8],
219    len: usize,
220    secret: &[u8],
221    secret_size: usize,
222) {
223    let nb_stripes_per_block = (secret_size - STRIPE_LEN) / SECRET_CONSUME_RATE;
224    let block_len = STRIPE_LEN * nb_stripes_per_block;
225    let nb_blocks = (len - 1) / block_len;
226    for n in 0..nb_blocks {
227        accumulate(acc, &input[n * block_len..], secret, nb_stripes_per_block);
228        scramble_acc(acc, &secret[secret_size - STRIPE_LEN..]);
229    }
230    let nb_stripes = ((len - 1) - (block_len * nb_blocks)) / STRIPE_LEN;
231    accumulate(acc, &input[nb_blocks * block_len..], secret, nb_stripes);
232    let p = len - STRIPE_LEN;
233    accumulate_512(
234        acc,
235        &input[p..],
236        &secret[secret_size - STRIPE_LEN - LASTACC_START..],
237    );
238}

240#[inline]
241fn mix2_accs(a0: u64, a1: u64, secret: &[u8]) -> u64 {
242    mul128_fold64(a0 ^ read_le64(secret, 0), a1 ^ read_le64(secret, 8))
243}

245fn merge_accs(acc: &[u64; 8], secret: &[u8], start: u64) -> u64 {
246    let mut result64 = start;
247    for i in 0..4 {
248        result64 = result64.wrapping_add(mix2_accs(acc[2 * i], acc[2 * i + 1], &secret[16 * i..]));
249    }
250    avalanche3(result64)
251}
252
253fn finalize_long(acc: &[u64; 8], secret: &[u8], len: u64) -> u64 {
254    merge_accs(acc, &secret[SECRET_MERGE_START..], len.wrapping_mul(P64_1))
255}

257fn xxh3_64(data: &[u8]) -> u64 {
258    let len = data.len();
259    let secret = &K3SECRET;
260    let seed = 0u64;
261    if len <= 16 {
262        len_0to16(data, len, secret, seed)
263    } else if len <= 128 {
264        len_17to128(data, len, secret, seed)
265    } else if len <= MIDSIZE_MAX {
266        len_129to240(data, len, secret, seed)
267    } else {
268        let mut acc = INIT_ACC;
269        hash_long_internal_loop(&mut acc, data, len, secret, secret.len());
270        finalize_long(&acc, secret, len as u64)
271    }
272}
