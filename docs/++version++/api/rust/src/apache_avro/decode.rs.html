45#[inline]
46pub(crate) fn decode_len<R: Read>(reader: &mut R) -> AvroResult<usize> {
47    let len = zag_i64(reader)?;
48    safe_len(usize::try_from(len).map_err(|e| Details::ConvertI64ToUsize(e, len))?)
49}

51/// Decode the length of a sequence.
52///
53/// Maps and arrays are 0-terminated, 0i64 is also encoded as 0 in Avro reading a length of 0 means
54/// the end of the map or array.
55fn decode_seq_len<R: Read>(reader: &mut R) -> AvroResult<usize> {
56    let raw_len = zag_i64(reader)?;
57    safe_len(
58        usize::try_from(match raw_len.cmp(&0) {
59            std::cmp::Ordering::Equal => return Ok(0),
60            std::cmp::Ordering::Less => {
61                let _size = zag_i64(reader)?;
62                raw_len.checked_neg().ok_or(Details::IntegerOverflow)?
63            }
64            std::cmp::Ordering::Greater => raw_len,
65        })
66        .map_err(|e| Details::ConvertI64ToUsize(e, raw_len))?,
67    )
68}

70/// Per-datum decoding state.
71///
72/// Tracks the cumulative number of bytes allocated on behalf of a single
73/// datum, so that nested collections cannot multiply the allocation budget:
74/// every allocation performed while decoding one datum is debited from a
75/// shared budget of [`max_allocation_bytes`] bytes, instead of each
76/// collection only being checked in isolation.
77#[derive(Debug)]
78pub(crate) struct DecodeContext {
79    /// Bytes still available for allocations while decoding the current datum.
80    remaining_budget: usize,
81    /// Current recursion depth of `decode_internal`.
82    depth: usize,
83}
84
85impl DecodeContext {
86    /// Create a new context.
87    ///
88    /// This should only be done when a new datum is being decoded, never during the decoding.
89    pub(crate) fn new() -> Self {
90        Self {
91            remaining_budget: max_allocation_bytes(DEFAULT_MAX_ALLOCATION_BYTES),
92            depth: 0,
93        }
94    }
95
96    /// Debit `bytes` from the per-datum allocation budget
97    ///
98    /// # Errors
99    /// `Details::MemoryAllocation` if the maximum budget is exceeded.
100    fn debit_bytes(&mut self, bytes: usize) -> AvroResult<()> {
101        match self.remaining_budget.checked_sub(bytes) {
102            Some(remaining) => {
103                self.remaining_budget = remaining;
104                Ok(())
105            }
106            None => Err(Details::MemoryAllocation {
107                desired: Some(bytes),
108                maximum: max_allocation_bytes(DEFAULT_MAX_ALLOCATION_BYTES),
109            }
110            .into()),
111        }
112    }
113
114    /// Debit the amount of bytes for `n` items of `T`.
115    ///
116    /// # Errors
117    /// `Details::MemoryAllocation` if the maximum budget is exceeded.
118    fn debit_items<T>(&mut self, n: usize) -> AvroResult<()> {
119        let bytes = n
120            .checked_mul(size_of::<T>())
121            .ok_or(Details::IntegerOverflow)?;
122        self.debit_bytes(bytes)
123    }
124
125    /// Track one level of decoding recursion, erroring once the configured
126    /// maximum depth is exceeded.
127    fn enter(&mut self) -> AvroResult<()> {
128        self.depth += 1;
129        let maximum = decode_recursion_limit();
130        if self.depth > maximum {
131            Err(Details::DecodeRecursionLimit { maximum }.into())
132        } else {
133            Ok(())
134        }
135    }
136
137    /// Leave one level of decoding recursion.
138    fn leave(&mut self) {
139        self.depth -= 1;
140    }
141}

143/// Decode a `Value` from avro format given its `Schema`.
144pub fn decode<R: Read>(schema: &Schema, reader: &mut R) -> AvroResult<Value> {
145    let rs = ResolvedSchema::try_from(schema)?;
146    decode_internal(
147        schema,
148        rs.get_names(),
149        None,
150        reader,
151        &mut DecodeContext::new(),
152    )
153}

155pub(crate) fn decode_internal<R: Read, S: Borrow<Schema>>(
156    schema: &Schema,
157    names: &HashMap<Name, S>,
158    enclosing_namespace: NamespaceRef,
159    reader: &mut R,
160    ctx: &mut DecodeContext,
161) -> AvroResult<Value> {
162    ctx.enter()?;
163    let value = decode_internal_body(schema, names, enclosing_namespace, reader, ctx);
164    ctx.leave();
165    value
166}
167
168fn decode_internal_body<R: Read, S: Borrow<Schema>>(
169    schema: &Schema,
170    names: &HashMap<Name, S>,
171    enclosing_namespace: NamespaceRef,
172    reader: &mut R,
173    ctx: &mut DecodeContext,
174) -> AvroResult<Value> {
175    match schema {
176        Schema::Null => Ok(Value::Null),
177        Schema::Boolean => {
178            let mut buf = [0u8; 1];
179            reader
180                .read_exact(&mut buf[..])
181                .map_err(Details::ReadBoolean)?;
182            match buf[0] {
183                0u8 => Ok(Value::Boolean(false)),
184                1u8 => Ok(Value::Boolean(true)),
185                _ => Err(Details::BoolValue(buf[0]).into()),
186            }
187        }
188        Schema::Decimal(DecimalSchema { inner, .. }) => match inner {
189            InnerDecimalSchema::Fixed(fixed) => {
190                match decode_internal(
191                    &Schema::Fixed(fixed.copy_only_size()),
192                    names,
193                    enclosing_namespace,
194                    reader,
195                    ctx,
196                )? {
197                    Value::Fixed(_, bytes) => Ok(Value::Decimal(Decimal::new(bytes)?)),
198                    value => Err(Details::FixedValue(value).into()),
199                }
200            }
201            InnerDecimalSchema::Bytes => {
202                match decode_internal(&Schema::Bytes, names, enclosing_namespace, reader, ctx)? {
203                    Value::Bytes(bytes) => Ok(Value::Decimal(Decimal::new(bytes)?)),
204                    value => Err(Details::BytesValue(value).into()),
205                }
206            }
207        },
208        Schema::BigDecimal => {
209            match decode_internal(&Schema::Bytes, names, enclosing_namespace, reader, ctx)? {
210                Value::Bytes(bytes) => deserialize_big_decimal(&bytes).map(Value::BigDecimal),
211                value => Err(Details::BytesValue(value).into()),
212            }
213        }
214        Schema::Uuid(UuidSchema::String) => {
215            let len = decode_len(reader)?;
216            if len <= uuid::fmt::Urn::LENGTH {
217                let mut buf = [0u8; uuid::fmt::Urn::LENGTH];
218                reader
219                    .read_exact(&mut buf[..len])
220                    .map_err(Details::ReadString)?;
221                let uuid = Uuid::try_parse_ascii(&buf[..len]).map_err(Details::ConvertStrToUuid)?;
222                Ok(Value::Uuid(uuid))
223            } else {
224                Err(Details::ConvertStringToUuid(uuid::fmt::Urn::LENGTH, len).into())
225            }
226        }
227        Schema::Uuid(UuidSchema::Bytes) => {
228            let len = decode_len(reader)?;
229            if len == 16 {
230                let mut buf = [0u8; 16];
231                reader.read_exact(&mut buf).map_err(Details::ReadBytes)?;
232                let uuid = Uuid::from_slice(&buf).map_err(Details::ConvertSliceToUuid)?;
233                Ok(Value::Uuid(uuid))
234            } else {
235                Err(Details::ConvertBytesToUuid(len).into())
236            }
237        }
238        Schema::Uuid(UuidSchema::Fixed(fixed)) => {
239            if fixed.size == 16 {
240                let mut buf = [0u8; 16];
241                reader
242                    .read_exact(&mut buf)
243                    .map_err(|e| Details::ReadFixed(e, 16))?;
244                let uuid = Uuid::from_slice(&buf).map_err(Details::ConvertSliceToUuid)?;
245                Ok(Value::Uuid(uuid))
246            } else {
247                Err(Details::ConvertFixedToUuid(fixed.size).into())
248            }
249        }
250        Schema::Int => decode_int(reader),
251        Schema::Date => zag_i32(reader).map(Value::Date),
252        Schema::TimeMillis => zag_i32(reader).map(Value::TimeMillis),
253        Schema::Long => decode_long(reader),
254        Schema::TimeMicros => zag_i64(reader).map(Value::TimeMicros),
255        Schema::TimestampMillis => zag_i64(reader).map(Value::TimestampMillis),
256        Schema::TimestampMicros => zag_i64(reader).map(Value::TimestampMicros),
257        Schema::TimestampNanos => zag_i64(reader).map(Value::TimestampNanos),
258        Schema::LocalTimestampMillis => zag_i64(reader).map(Value::LocalTimestampMillis),
259        Schema::LocalTimestampMicros => zag_i64(reader).map(Value::LocalTimestampMicros),
260        Schema::LocalTimestampNanos => zag_i64(reader).map(Value::LocalTimestampNanos),
261        Schema::Duration(fixed_schema) => {
262            if fixed_schema.size == 12 {
263                let mut buf = [0u8; 12];
264                reader.read_exact(&mut buf).map_err(Details::ReadDuration)?;
265                Ok(Value::Duration(Duration::from(buf)))
266            } else {
267                Err(Details::CompareFixedSizes {
268                    size: 12,
269                    n: fixed_schema.size,
270                }
271                .into())
272            }
273        }
274        Schema::Float => {
275            let mut buf = [0u8; std::mem::size_of::<f32>()];
276            reader
277                .read_exact(&mut buf[..])
278                .map_err(Details::ReadFloat)?;
279            Ok(Value::Float(f32::from_le_bytes(buf)))
280        }
281        Schema::Double => {
282            let mut buf = [0u8; std::mem::size_of::<f64>()];
283            reader
284                .read_exact(&mut buf[..])
285                .map_err(Details::ReadDouble)?;
286            Ok(Value::Double(f64::from_le_bytes(buf)))
287        }
288        Schema::Bytes => {
289            let len = decode_len(reader)?;
290            ctx.debit_bytes(len)?;
291            let mut buf = vec![0u8; len];
292            reader.read_exact(&mut buf).map_err(Details::ReadBytes)?;
293            Ok(Value::Bytes(buf))
294        }
295        Schema::String => {
296            let len = decode_len(reader)?;
297            ctx.debit_bytes(len)?;
298            let mut buf = vec![0u8; len];
299            reader.read_exact(&mut buf).map_err(Details::ReadString)?;
300            Ok(Value::String(
301                String::from_utf8(buf).map_err(Details::ConvertToUtf8)?,
302            ))
303        }
304        Schema::Fixed(FixedSchema { size, .. }) => {
305            ctx.debit_bytes(*size)?;
306            let mut buf = vec![0u8; *size];
307            reader
308                .read_exact(&mut buf)
309                .map_err(|e| Details::ReadFixed(e, *size))?;
310            Ok(Value::Fixed(*size, buf))
311        }
312        Schema::Array(inner) => {
313            let mut items = Vec::new();
314
315            loop {
316                let len = decode_seq_len(reader)?;
317                if len == 0 {
318                    break;
319                }
320
321                // Check that the Vec won't grow past the max allocation size
322                let total = items
323                    .len()
324                    .checked_add(len)
325                    .ok_or(Details::IntegerOverflow)?;
326                safe_collection_len::<Value>(total)?;
327                ctx.debit_items::<Value>(len)?;
328                // Use reserve_exact as reserve can allocate more than needed defeating the purpose
329                // of the previous check
330                items.reserve_exact(len);
331                for _ in 0..len {
332                    items.push(decode_internal(
333                        &inner.items,
334                        names,
335                        enclosing_namespace,
336                        reader,
337                        ctx,
338                    )?);
339                }
340            }
341
342            Ok(Value::Array(items))
343        }
344        Schema::Map(inner) => {
345            let mut items = HashMap::new();
346
347            loop {
348                let len = decode_seq_len(reader)?;
349                if len == 0 {
350                    break;
351                }
352
353                // Check that the HashMap won't grow past the max allocation size. This is less
354                // precise than the Vec check above as HashMap allocates in buckets and doesn't have
355                // a reserve_exact
356                let total = items
357                    .len()
358                    .checked_add(len)
359                    .ok_or(Details::IntegerOverflow)?;
360                safe_collection_len::<(String, Value)>(total)?;
361                ctx.debit_items::<(String, Value)>(len)?;
362
363                items.reserve(len);
364                for _ in 0..len {
365                    match decode_internal(&Schema::String, names, enclosing_namespace, reader, ctx)?
366                    {
367                        Value::String(key) => {
368                            let value = decode_internal(
369                                &inner.types,
370                                names,
371                                enclosing_namespace,
372                                reader,
373                                ctx,
374                            )?;
375                            items.insert(key, value);
376                        }
377                        value => return Err(Details::MapKeyType(value.into()).into()),
378                    }
379                }
380            }
381
382            Ok(Value::Map(items))
383        }
384        Schema::Union(inner) => {
385            let index = zag_i64(reader)?;
386            let variants = inner.variants();
387            let variant = variants
388                .get(usize::try_from(index).map_err(|e| Details::ConvertI64ToUsize(e, index))?)
389                .ok_or(Details::GetUnionVariant {
390                    index,
391                    num_variants: variants.len(),
392                })?;
393            let value = decode_internal(variant, names, enclosing_namespace, reader, ctx)?;
394            Ok(Value::Union(index as u32, Box::new(value)))
395        }
396        Schema::Record(RecordSchema { name, fields, .. }) => {
397            let fully_qualified_name = name.fully_qualified_name(enclosing_namespace);
398            ctx.debit_items::<(String, Value)>(fields.len())?;
399            // Benchmarks indicate ~10% improvement using this method.
400            let mut items = Vec::with_capacity(fields.len());
401            for field in fields {
402                ctx.debit_bytes(field.name.len())?;
403                // TODO: This clone is also expensive. See if we can do away with it...
404                items.push((
405                    field.name.clone(),
406                    decode_internal(
407                        &field.schema,
408                        names,
409                        fully_qualified_name.namespace(),
410                        reader,
411                        ctx,
412                    )?,
413                ));
414            }
415            Ok(Value::Record(items))
416        }
417        Schema::Enum(EnumSchema { symbols, .. }) => {
418            Ok(if let Value::Int(raw_index) = decode_int(reader)? {
419                let index = usize::try_from(raw_index)
420                    .map_err(|e| Details::ConvertI32ToUsize(e, raw_index))?;
421                if (0..symbols.len()).contains(&index) {
422                    // Cloning the symbol allocates without consuming wire
423                    // bytes, so it counts against the per-datum budget.
424                    ctx.debit_bytes(symbols[index].len())?;
425                    let symbol = symbols[index].clone();
426                    Value::Enum(raw_index as u32, symbol)
427                } else {
428                    return Err(Details::GetEnumValue {
429                        index,
430                        nsymbols: symbols.len(),
431                    }
432                    .into());
433                }
434            } else {
435                return Err(Details::GetEnumUnknownIndexValue.into());
436            })
437        }
438        Schema::Ref { name } => {
439            let fully_qualified_name = name.fully_qualified_name(enclosing_namespace);
440            if let Some(resolved) = names.get(&fully_qualified_name) {
441                decode_internal(
442                    resolved.borrow(),
443                    names,
444                    fully_qualified_name.namespace(),
445                    reader,
446                    ctx,
447                )
448            } else {
449                Err(Details::SchemaResolutionError(fully_qualified_name.into_owned()).into())
450            }
451        }
452    }

562    #[test]
563    fn avro_rs_639_test_nested_collections_share_one_allocation_budget() -> TestResult {
564        use crate::util::{DEFAULT_MAX_ALLOCATION_BYTES, max_allocation_bytes};
565
566        // Each inner array<null> block passes the per-collection check on its
567        // own, but the shared per-datum budget must reject the cumulative
568        // total: elements of type null cost zero wire bytes, so without a
569        // cumulative budget a handful of ~10-byte inner arrays would pin an
570        // unbounded multiple of the allocation limit in memory at once.
571        let budget = max_allocation_bytes(DEFAULT_MAX_ALLOCATION_BYTES);
572        let inner_count = (budget / 2) / size_of::<Value>() + 1;
573
574        let inner_arrays_count = 2;
575        let mut payload = Vec::new();
576        // Outer array: a single block declaring two inner arrays.
577        crate::util::zig_i64(inner_arrays_count, &mut payload)?;
578        for _ in 0..inner_arrays_count {
579            payload.extend(create_block(inner_count as i64));
580        }
581        // Outer array terminator.
582        payload.push(0x00);
583
584        let result = decode(
585            &Schema::array(Schema::array(Schema::Null).build()).build(),
586            &mut payload.as_slice(),
587        );
588
589        assert!(
590            result.is_err(),
591            "nested collections must share one allocation budget, got {result:?}"
592        );
593        let details = result.unwrap_err().into_details();
594        assert!(
595            matches!(details, Details::MemoryAllocation { .. }),
596            "Expected memory allocation error, got: {details:?}"
597        );
598
599        Ok(())
600    }
