1use super::reject_attrs;
2use crate::util::prelude::*;
3use syn::Expr;
4
5/// At the time of this writing `Rust` doesn't support calling closures in `const`
6/// contexts. This doesn't compile:
7///
8/// ```compile_fail
9/// const { (|| {})() }
10/// ```
11///
12/// However, we need this to work because we need to wrap user-supplied expressions
13/// with closures. It gives us a guarantee that the user-supplied expression can't
14/// break out of the surrounding scope via a `return`, `break`, or `continue` inside
15/// of it.
16///
17/// Therefore, we use this function to allow very simple expressions to still be
18/// embedded in the surrounding context while recommending to delegate to a function
19/// call if more complex expression is required.
20pub(crate) fn require_embeddable_const_expr(expr: &Expr) -> Result {
21    use require_embeddable_const_expr as recurse;
22
23    fn recurse_block(block: &syn::Block) -> Result {
24        let stmt = match block.stmts.as_slice() {
25            [stmt] => stmt,
26            [] => return Ok(()),
27            _ => bail!(
28                &block,
29                "only a single expression in a block is allowed in this position"
30            ),
31        };
32
33        match stmt {
34            syn::Stmt::Expr(expr, None) => recurse(expr),
35            _ => bail!(
36                &stmt,
37                "this kind of statement is not allowed in this position; \
38                only a single expression is allowed in a block is allowed here"
39            ),
40        }
41    }
42
43    match expr {
44        Expr::Array(arr) => {
45            reject_attrs(&arr.attrs)?;
46            arr.elems.iter().try_for_each(recurse)?;
47        }
48        Expr::Binary(binary) => {
49            reject_attrs(&binary.attrs)?;
50            recurse(&binary.left)?;
51            recurse(&binary.right)?;
52        }
53        Expr::Block(block) => {
54            reject_attrs(&block.attrs)?;
55            recurse_block(&block.block)?;
56        }
57        Expr::Call(call) => {
58            reject_attrs(&call.attrs)?;
59            recurse(&call.func)?;
60            call.args.iter().try_for_each(recurse)?;
61        }
62        Expr::Cast(cast) => {
63            reject_attrs(&cast.attrs)?;
64            recurse(&cast.expr)?;
65        }
66        Expr::Field(field) => {
67            reject_attrs(&field.attrs)?;
68            recurse(&field.base)?;
69        }
70        Expr::Group(group) => {
71            reject_attrs(&group.attrs)?;
72            recurse(&group.expr)?;
73        }
74        Expr::If(if_expr) => {
75            reject_attrs(&if_expr.attrs)?;
76            recurse(&if_expr.cond)?;
77            recurse_block(&if_expr.then_branch)?;
78            if let Some((_else, else_branch)) = &if_expr.else_branch {
79                recurse(else_branch)?;
80            }
81        }
82        Expr::Const(const_block) => {
83            reject_attrs(&const_block.attrs)?;
84            recurse_block(&const_block.block)?;
85        }
86        Expr::Index(index) => {
87            reject_attrs(&index.attrs)?;
88            recurse(&index.expr)?;
89            recurse(&index.index)?;
90        }
91        Expr::Infer(infer) => reject_attrs(&infer.attrs)?,
92        Expr::Lit(lit) => reject_attrs(&lit.attrs)?,
93        Expr::Loop(loop_expr) => {
94            reject_attrs(&loop_expr.attrs)?;
95            recurse_block(&loop_expr.body)?;
96        }
97        Expr::Match(expr_match) => {
98            reject_attrs(&expr_match.attrs)?;
99            recurse(&expr_match.expr)?;
100            expr_match.arms.iter().try_for_each(|arm| {
101                reject_attrs(&arm.attrs)?;
102
103                if let syn::Pat::Guard(guard) = &arm.pat {
104                    recurse(&guard.guard)?;
105                }
106
107                recurse(&arm.body)
108            })?;
109        }
110        Expr::MethodCall(method_call) => {
111            reject_attrs(&method_call.attrs)?;
112            recurse(&method_call.receiver)?;
113            method_call.args.iter().try_for_each(recurse)?;
114        }
115        Expr::Paren(paren) => {
116            reject_attrs(&paren.attrs)?;
117            recurse(&paren.expr)?;
118        }
119        Expr::Path(path) => reject_attrs(&path.attrs)?,
120        Expr::Range(range) => {
121            reject_attrs(&range.attrs)?;
122            if let Some(start) = &range.start {
123                recurse(start)?;
124            }
125            if let Some(end) = &range.end {
126                recurse(end)?;
127            }
128        }
129
130        Expr::Reference(reference) => {
131            reject_attrs(&reference.attrs)?;
132            recurse(&reference.expr)?;
133        }
134        Expr::Repeat(repeat) => {
135            reject_attrs(&repeat.attrs)?;
136            recurse(&repeat.expr)?;
137            recurse(&repeat.len)?;
138        }
139        Expr::Struct(struct_expr) => {
140            reject_attrs(&struct_expr.attrs)?;
141            struct_expr.fields.iter().try_for_each(|field| {
142                reject_attrs(&field.attrs)?;
143                recurse(&field.expr)
144            })?;
145            if let Some(rest) = &struct_expr.rest {
146                recurse(rest)?;
147            }
148        }
149        Expr::Tuple(tuple) => {
150            reject_attrs(&tuple.attrs)?;
151            tuple.elems.iter().try_for_each(recurse)?;
152        }
153        Expr::Unary(unary) => {
154            reject_attrs(&unary.attrs)?;
155            recurse(&unary.expr)?;
156        }
157        Expr::Unsafe(unsafe_expr) => {
158            reject_attrs(&unsafe_expr.attrs)?;
159            recurse_block(&unsafe_expr.block)?;
160        }
161        Expr::While(while_expr) => {
162            reject_attrs(&while_expr.attrs)?;
163            recurse(&while_expr.cond)?;
164            recurse_block(&while_expr.body)?;
165        }
166        _ => {
167            bail!(
168                &expr,
169                "this kind of expression is not allowed in this position; \
170                if you need to use a complex expression such as this then \
171                move it into a separate `const fn` and call that function \
172                here instead"
173            )
174        }
175    }
176
177    Ok(())
178}
